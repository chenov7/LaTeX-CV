% Section title (blue)
\newcommand{\cvsection}[1]{%
    \vspace{1.2em}
    {\color{primary}\large\bfseries #1}
    \vspace{0.6em}
    \par
}

% Section title for right column (more spacing)
\newcommand{\cvsectionright}[1]{%
    \vspace{2.5em}
    {\color{primary}\large\bfseries #1}
    \vspace{0.6em}
    \par
}

% Subsection (experience/formation title)
\newcommand{\cvsubsection}[2]{%
    \textbf{#1} {\color{primary}-- #2}
    \par
}

% Date in gray
\newcommand{\cvdate}[1]{%
    {\color{lightgray}\small #1}
    \par
}

% Description text
\newcommand{\cvdesc}[1]{%
    {\color{darkgray}\small #1}
    \par
}

% Skill bullet point (small 3pt square)
% #2 = main skill (darkgray), #1 = optional sub-skills (lightgray)
\newcommand{\cvskill}[2][]{%
    {\small\raisebox{0.3ex}{\color{mediumgray}\rule{3pt}{3pt}}\hspace{0.5em}{\color{darkgray}#2}%
    \if\relax\detokenize{#1}\relax\else{\color{lightgray} ~- #1}\fi}%
    \par
}

% Contact item (icon with fixed width for alignment)
\newcommand{\cvcontact}[2]{%
    {\color{lightgray}\makebox[1.2em][l]{#1}#2}
    \par\vspace{0.2em}
}
